<script>

let currentRates = {};
let historyDays = 7;

document.getElementById("year").textContent =
  new Date().getFullYear();

const currencyIcons = {
  USD: "🇺🇸",
  EUR: "🇪🇺",
  GBP: "🇬🇧",
  USDT: "₮"
};

function showSection(section, button) {

  document.querySelectorAll(".section").forEach(item => {
    item.classList.remove("active");
  });

  document.getElementById(section).classList.add("active");

  document.querySelectorAll(".nav-item").forEach(item => {
    item.classList.remove("active");
  });

  button.classList.add("active");

  if (section === "history") {
    loadHistory();
  }

  window.scrollTo({top: 0, behavior: "smooth"});
}

async function loadRates() {

  const list = document.getElementById("ratesList");

  try {

    const response = await fetch("/api/rates", {
      cache: "no-store"
    });

    if (!response.ok) {
      throw new Error("Rate request failed");
    }

    const data = await response.json();

    if (!data.success || !data.rates) {
      throw new Error("Invalid rate data");
    }

    currentRates = data.rates;

    const currencies = ["USD", "EUR", "GBP", "USDT"];

    list.innerHTML = currencies.map(currency => {

      const rate = currentRates[currency];

      if (!rate) {
        return `
          <div class="rate-card">
            <div class="currency-name">
              ${currencyIcons[currency]} ${currency}
            </div>
            <p style="color:#999;margin-top:12px">
              Rate unavailable
            </p>
          </div>
        `;
      }

      return `
        <div class="rate-card">

          <div class="rate-top">
            <div class="currency-name">
              <span class="currency-icon">
                ${currencyIcons[currency]}
              </span>
              ${currency}
            </div>

            <span class="currency-code">
              LIVE
            </span>
          </div>

          <div class="rate-bottom">

            <div class="rate-box">
              <div class="rate-label">BUY RATE</div>
              <div class="rate-price buy-price">
                ₦${Number(rate.buy).toLocaleString("en-NG")}
              </div>
            </div>

            <div class="rate-box">
              <div class="rate-label">SELL RATE</div>
              <div class="rate-price sell-price">
                ₦${Number(rate.sell).toLocaleString("en-NG")}
              </div>
            </div>

          </div>
        </div>
      `;

    }).join("");

    document.getElementById("lastUpdated").textContent =
      "Last updated: " + new Date().toLocaleString();

  } catch (error) {

    list.textContent =
      "Unable to load rates. Please refresh the page.";

    document.getElementById("lastUpdated").textContent =
      "Connection error";

  }

}

function calculate() {

  const currency =
    document.getElementById("calculatorCurrency").value;

  const amount =
    Number(document.getElementById("amount").value);

  const type =
    document.getElementById("rateType").value;

  const rate = currentRates[currency];

  if (!rate) {
    alert("Rates are not available yet.");
    return;
  }

  if (!Number.isFinite(amount) || amount <= 0) {
    alert("Please enter a valid amount.");
    return;
  }

  const value = amount * Number(rate[type]);

  document.getElementById("calculationResult").textContent =
    "₦" + value.toLocaleString("en-NG", {
      minimumFractionDigits: 2,
      maximumFractionDigits: 2
    });

}

document.getElementById("historyCurrency")

function changeDays(days) {

  historyDays = days;

  document.getElementById("days7")
    .classList.toggle("active", days === 7);

  document.getElementById("days30")
    .classList.toggle("active", days === 30);

  loadHistory();

}

async function loadHistory() {

  const currency =
    document.getElementById("historyCurrency").value;

  const status =
    document.getElementById("chartStatus");

  status.textContent = "Loading history...";

  try {

    const response = await fetch(
      `/api/history?currency=${currency}&days=${historyDays}`,
      {cache: "no-store"}
    );

    if (!response.ok) {
      throw new Error("History request failed");
    }

    const data = await response.json();

    if (!data.success || !Array.isArray(data.history)) {
      throw new Error("Invalid history data");
    }

    drawHistoryChart(data.history);

  } catch (error) {

    status.textContent =
      "Unable to load history. Please try again.";

    drawHistoryChart([]);

  }

}

function drawHistoryChart(rows) {

  const canvas =
    document.getElementById("historyChart");

  const status =
    document.getElementById("chartStatus");

  const width = canvas.clientWidth;
  const height = canvas.clientHeight;

  const ratio = window.devicePixelRatio || 1;

  canvas.width = width * ratio;
  canvas.height = height * ratio;

  const ctx = canvas.getContext("2d");

  ctx.setTransform(ratio, 0, 0, ratio, 0, 0);

  ctx.clearRect(0, 0, width, height);

  if (!rows || rows.length === 0) {
    status.textContent =
      "History will appear as rates are updated.";
    return;
  }

  const padding = {
    left: 48,
    right: 12,
    top: 20,
    bottom: 30
  };

  const chartWidth =
    width - padding.left - padding.right;

  const chartHeight =
    height - padding.top - padding.bottom;

  const values = rows.flatMap(row => [
    Number(row.buy),
    Number(row.sell)
  ]).filter(Number.isFinite);

  if (!values.length) {
    status.textContent = "No valid history values.";
    return;
  }

  let min = Math.min(...values);
  let max = Math.max(...values);

  if (min === max) {
    min -= 1;
    max += 1;
  }

  const extra = (max - min) * 0.15;

  min -= extra;
  max += extra;

  function x(index) {
    return padding.left +
      (rows.length === 1
        ? chartWidth / 2
        : index * chartWidth / (rows.length - 1));
  }

  function y(value) {
    return padding.top +
      (max - value) / (max - min) * chartHeight;
  }

  ctx.font = "10px Arial";
  ctx.fillStyle = "#aaa";
  ctx.strokeStyle = "#333";
  ctx.lineWidth = 1;

  for (let i = 0; i <= 4; i++) {

    const value = max - (max - min) * i / 4;
    const yy = y(value);

    ctx.beginPath();
    ctx.moveTo(padding.left, yy);
    ctx.lineTo(width - padding.right, yy);
    ctx.stroke();

    ctx.fillText(
      Math.round(value).toLocaleString(),
      2,
      yy + 4
    );

  }

  function drawLine(key, color) {

    ctx.beginPath();
    ctx.strokeStyle = color;
    ctx.lineWidth = 2;

    let started = false;

    rows.forEach((row, index) => {

      const value = Number(row[key]);

      if (!Number.isFinite(value)) return;

      if (!started) {
        ctx.moveTo(x(index), y(value));
        started = true;
      } else {
        ctx.lineTo(x(index), y(value));
      }

    });

    ctx.stroke();

    rows.forEach((row, index) => {

      const value = Number(row[key]);

      if (!Number.isFinite(value)) return;

      ctx.beginPath();
      ctx.fillStyle = color;
      ctx.arc(x(index), y(value), 3, 0, Math.PI * 2);
      ctx.fill();

    });

  }

  drawLine("buy", "#35c879");
  drawLine("sell", "#d8ad45");

  status.textContent =
    rows.length + " recorded rate updates shown.";

}

async function shareWebsite() {

  const shareData = {
    title: "Rasheed_Fx Rates",
    text: "Check the latest exchange rates on Rasheed_Fx.",
    url: window.location.href
  };

  try {

    if (navigator.share) {
      await navigator.share(shareData);
    } else {
      await navigator.clipboard.writeText(window.location.href);
      alert("Website link copied.");
    }

  } catch (error) {
    // Sharing was cancelled or unavailable.
  }

}
